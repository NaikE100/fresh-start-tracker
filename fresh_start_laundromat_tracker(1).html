Supabase SQL Table Schema Setup

Run this SQL code inside your Supabase project's SQL Editor tab to create the required daily_logs table: 

CREATE TABLE IF NOT EXISTS daily_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_code(),
    date DATE UNIQUE NOT NULL,
    wash_cycles INT DEFAULT 0,
    dryer_cycles INT DEFAULT 0,
    electricity_kwh NUMERIC DEFAULT 0,
    electricity_rate NUMERIC DEFAULT 0,
    water_kl NUMERIC DEFAULT 0,
    water_rate NUMERIC DEFAULT 0,
    direct_expenses NUMERIC DEFAULT 0,
    daily_income NUMERIC DEFAULT 0,
    labor_cost NUMERIC DEFAULT 0,
    transport_expense NUMERIC DEFAULT 0,
    total_expenses NUMERIC DEFAULT 0,
    net_profit NUMERIC DEFAULT 0,
    employees JSONB DEFAULT '[]'::jsonb,
    trips JSONB DEFAULT '[]'::jsonb,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable Row Level Security (RLS)
ALTER TABLE daily_logs ENABLE ROW LEVEL SECURITY;

-- Allow public read & write access for application operations
CREATE POLICY "Allow public read access" ON daily_logs FOR SELECT USING (true);
CREATE POLICY "Allow public insert and update access" ON daily_logs FOR ALL USING (true);
